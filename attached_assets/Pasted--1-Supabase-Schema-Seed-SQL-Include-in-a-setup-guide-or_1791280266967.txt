#### 1. Supabase Schema & Seed SQL (Include in a setup guide or migration script)
```sql
-- Create banks table
CREATE TABLE public.banks (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  name TEXT NOT NULL,
  logo_url TEXT,
  created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL
);

-- Create loan products table
CREATE TABLE public.loan_products (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  bank_id UUID REFERENCES public.banks(id) ON DELETE CASCADE,
  loan_type TEXT NOT NULL, -- 'Personal', 'Car', 'Housing', 'Education'
  min_interest_rate NUMERIC(5,2) NOT NULL,
  max_interest_rate NUMERIC(5,2),
  max_tenure_months INT NOT NULL,
  max_loan_amount BIGINT NOT NULL,
  processing_fee_desc TEXT,
  features TEXT[],
  created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL
);

-- Create saved comparisons / applications table
CREATE TABLE public.loan_applications (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  user_name TEXT NOT NULL,
  email TEXT NOT NULL,
  phone TEXT NOT NULL,
  loan_product_id UUID REFERENCES public.loan_products(id),
  requested_amount NUMERIC NOT NULL,
  tenure_months INT NOT NULL,
  estimated_emi NUMERIC NOT NULL,
  status TEXT DEFAULT 'Pending',
  created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL
);

-- Enable RLS and create public read policies
ALTER TABLE public.banks ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.loan_products ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.loan_applications ENABLE ROW LEVEL SECURITY;

CREATE POLICY "Allow public read banks" ON public.banks FOR SELECT USING (true);
CREATE POLICY "Allow public read loan_products" ON public.loan_products FOR SELECT USING (true);
CREATE POLICY "Allow public insert loan_applications" ON public.loan_applications FOR INSERT WITH CHECK (true);
CREATE POLICY "Allow public read own applications" ON public.loan_applications FOR SELECT USING (true);
